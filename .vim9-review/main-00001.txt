/// Keep history controls inside the window while avoiding an oversized empty
/// modal for a short result set. The list receives only the remaining height.
fn history_picker_panel_size(window: Size, visible_results: usize) -> Size {
    let list_height = (visible_results.max(1) as f32 * 44.0).min(300.0);
    Size::new(
        (window.width - 24.0).clamp(1.0, 560.0),
        (window.height - 24.0).clamp(1.0, list_height + 160.0),
    )
}

/// Bound workflow panels to the window while letting their contents scroll.
fn workflow_panel_size(window: Size, requested_height: f32) -> Size {
    Size::new(
        (window.width - 32.0).clamp(1.0, 560.0),
        (window.height - 32.0)
            .max(1.0)
            .min(requested_height.clamp(1.0, 480.0)),
    )
}

/// One real workflow argument input, shared with keyboard-routing regressions.
fn workflow_argument_input<'a, Renderer>(
    identity: workflow_picker::WorkflowArgsIdentity,
    index: usize,
    name: &str,
    value: &str,
) -> iced::widget::TextInput<'a, Message, iced::Theme, Renderer>
where
    Renderer: iced::advanced::text::Renderer<Font = iced::Font>,
{
    let input = iced::widget::TextInput::new(name, value)
        .on_input(move |value| Message::WorkflowArgInput(identity.clone(), index, value))
        // Enter must remain uncaptured so the shared physical-key latch can
        // reject an opener's repeats before the argument-stage handler runs.
        .size(13)
        .font(iced::Font::MONOSPACE);
    if index == 0 {
        input.id(WORKFLOW_ARG_INPUT_ID.clone())
    } else {
        input
    }
}

/// Keep block review inside the available window; compact layouts scroll the
/// heading and status together with the preview so explicit actions stay visible.
fn block_review_panel_size(window: Size) -> Size {
    Size::new(
        (window.width - 32.0).clamp(1.0, 880.0),
        (window.height - 48.0).max(1.0),
    )
}

fn block_review_scrolls_heading(panel: Size) -> bool {
    panel.width < 360.0 || panel.height < 300.0
}

/// Vertical chrome around the terminal area: the always-present tab bar plus
/// the bottom bar when the `bottom_bar` toggle is on.
fn chrome_height(font_size: f32, chrome_font_size: f32, bottom_bar: bool) -> f32 {
    ui_typography::row_height(chrome_font_size, TAB_BAR_H, 14.0)
        + if bottom_bar {
            ui_typography::row_height(font_size, STATUS_BAR_H, 4.0)
        } else {
            0.0
        }
}

const COMPANION_BODY_WIDTH: f32 = 200.0;
const COMPANION_HORIZONTAL_PADDING: f32 = 10.0;
const COMPANION_CARD_SPRITE_WIDTH: f32 = 104.0;
const COMPANION_CARD_DETAILS_MIN_WIDTH: f32 = 360.0;
const ORGANISM_STATUS_MIN_WIDTH: f32 = 640.0;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct OrganismLayout {
    glyph: bool,
    body: bool,
    explanation: bool,
}

impl OrganismLayout {
    fn presentable(self) -> bool {
        self.glyph || self.body
    }

    fn maintenance_eligible(self, focused: bool) -> bool {
        focused && self.presentable()
    }
}

/// Preserve the established 220-unit expanded minimum and 640-unit status
/// threshold. The native card may show details before the crowded status row
/// fits; its body and collapse control have priority. Iced logical units.
fn native_status_rows(enabled: bool, expanded: bool, cell_height: f32) -> usize {
    if !enabled {
        return 0;
    }
    if !expanded {
        return 1;
    }
    ((3.0 * cell_height + 28.0) / cell_height.max(1.0))
        .ceil()
        .clamp(3.0, 8.0) as usize
}

/// Card admission uses the same visible pane width as the terminal renderer.
fn organism_card_width(
    panes: &[PaneRect],
    active: usize,
    zoomed: bool,
    terminal_width: f32,
    padding: f32,
) -> f32 {
    panes
        .iter()
        .find(|pane| pane.session == active)
        .map_or(0.0, |pane| {
            let width = if zoomed {
                terminal_width
            } else {
                pane.rect.width
            };
            (width - padding - terminal_view::SCROLLBAR_WIDTH - 16.0).max(0.0)
        })
}

fn organism_host_layout(
    width: f32,
    enabled: bool,
    expanded: bool,
    bottom_bar: bool,
    block_mode: bool,
) -> OrganismLayout {
    if block_mode {
        let presentable = enabled && width.is_finite() && width >= 160.0;
        OrganismLayout {
            glyph: false,
            body: presentable,
            explanation: presentable && expanded && width >= COMPANION_CARD_DETAILS_MIN_WIDTH,
        }
    } else {
        organism_layout(width, enabled, expanded, bottom_bar)
    }
}

fn organism_layout(width: f32, enabled: bool, expanded: bool, bottom_bar: bool) -> OrganismLayout {
    let enabled = enabled && bottom_bar && width.is_finite();
    let glyph = enabled && width >= ORGANISM_STATUS_MIN_WIDTH;
    let body =
        enabled && expanded && width >= COMPANION_BODY_WIDTH + 2.0 * COMPANION_HORIZONTAL_PADDING;
    OrganismLayout {
        glyph,
        body,
        explanation: body && width >= COMPANION_CARD_DETAILS_MIN_WIDTH,
    }
}

/// A narrow collapsed card still has a way back, without admitting a live body
/// or hover observer while it is hidden. Configuration chrome only.
fn companion_narrow_reopen(width: f32, enabled: bool, expanded: bool, bottom_bar: bool) -> bool {
    !expanded
        && !organism_layout(width, enabled, false, bottom_bar).glyph
        && organism_layout(width, enabled, true, bottom_bar).body
}
