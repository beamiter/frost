        let section = |title: &str| -> Element<'_, Message> {
            text(title.to_string())
                .size(self.effective_font_size())
                .style(text::primary)
                .into()
        };
        let kb = |key: &str, desc: &str| -> Element<'_, Message> {
            row![
                container(
                    text(key.to_string())
                        .size(self.effective_font_size())
                        .font(iced::Font::MONOSPACE)
                )
                .width(Length::Fixed(190.0)),
                text(desc.to_string())
                    .size(self.effective_font_size())
                    .style(text::secondary),
            ]
            .spacing(8)
            .into()
        };
        // Every row whose chord is configurable reads the live binding table.
        // These were literals, so the one panel whose whole job is to teach
        // the keyboard showed frost's shipped defaults to a user who had
        // rebound them — and still showed a chord for a command they had
        // unbound. An unbound command now says so instead.
        let bound = |command_id: &str, desc: &str| -> Element<'_, Message> {
            let key = self
                .keybindings
                .shortcut_label(command_id)
                .unwrap_or_else(|| "(unbound)".to_string());

                command_ids
                    .iter()
                    .map(|command_id| {
                        self.keybindings
                            .shortcut_label(command_id)
                            .unwrap_or_else(|| "(unbound)".to_string())
                    })

    /// Top-right diagnostics overlay (F12): live grid / session /
    /// scrollback / Kitty-image / process-memory stats for the active session.
    fn debug_panel(&self) -> Element<'_, Message> {
        let stat = |label: &str, value: String| -> Element<'_, Message> {
            row![
                container(
                    text(label.to_string())
                        .size(self.effective_font_size())
                        .style(text::primary)
                )
                .width(Length::Fixed(110.0)),
                text(value)
                    .size(self.effective_font_size())
                    .font(iced::Font::MONOSPACE),
            ]
            .spacing(8)
            .into()
        };

        let mut lines = column![text("Diagnostics").size(self.effective_font_size())].spacing(3);
        let wakes = self.organism_wake_counts.snapshot();
        lines = lines
            .push(stat("Motion", format!("{:?} / {:?}", self.organism_motion_control.resolution(), self.organism_motion_control.source())))
            .push(stat("Organism wake", format!("live {} preview {} maintenance {}", wakes.live_animation, wakes.preview_animation, wakes.maintenance)))
            .push(stat("Expiry wake", format!("preview {}/{}/{} retreat {} completion {} attention {}", wakes.preview_boundary, wakes.preview_greeting, wakes.preview_cooldown, wakes.retreat_expiry, wakes.completion_expiry, wakes.live_attention)))
            .push(text("Delivered organism causes since launch; not frame or application-wide wake counts.").size(self.effective_font_size()));
        lines = lines
            .push(stat("Grid", format!("{}x{}", self.cols, self.rows)))
            .push(stat("Sessions", format!("{}", self.sessions.len())))
            .push(stat("Active", format!("#{}", self.active + 1)))
            .push(stat(
                "Split",
                if self.is_split() {
                    format!(
                        "{}/{} panes",
                        self.focused_pane_pos() + 1,
                        self.layout().leaf_count()
                    )
                } else {
                    "Single".to_string()
                },
            ));
        if let Some(sess) = self.sessions.get(self.active) {
            lines = lines
                .push(stat(
                    "Scrollback",
                    format!(
                        "{} / {}",
                        sess.terminal.scrollback_len(),
                        self.config.scrollback_lines
                    ),
                ))
                .push(stat(
                    "Scroll Off",
                    format!("{}", sess.terminal.scroll_offset),
                ))
                .push(stat(
                    "Kitty Imgs",
                    format!("{}", sess.terminal.kitty_graphics.image_count()),
                ))
                .push(stat(
                    "Kitty Mem",
